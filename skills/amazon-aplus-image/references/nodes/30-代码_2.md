```python
import json
from typing import Any


def safe_json_parse(value: Any) -> Any:
    if not isinstance(value, str):
        return value

    text = value.strip()

    try:
        return json.loads(text)
    except Exception:
        return value


async def main(args):
    if hasattr(args, "params"):
        params = args.params
    elif isinstance(args, dict) and "params" in args:
        params = args["params"]
    elif isinstance(args, dict):
        params = args
    else:
        params = {}

    # 取 images
    images = params.get("images", [])

    # 如果 images 是字串，先 parse
    images = safe_json_parse(images)

    # 情況 1：images 本身就是 list
    if isinstance(images, list) and len(images) > 0:
        first_item = images[0]

        # 如果 list 裡面直接就是 url 字串
        if isinstance(first_item, str) and first_item.startswith("http"):
            return {
                "image2url": first_item
            }

        # 如果 list 裡面是 dict，嘗試抓常見欄位
        if isinstance(first_item, dict):
            for key in ["url", "image_url", "file_url", "uri"]:
                val = first_item.get(key)
                if isinstance(val, str) and val.startswith("http"):
                    return {
                        "image2url": val
                    }

    # 情況 2：images 是 dict，且裡面包了一層 images
    if isinstance(images, dict):
        inner_images = images.get("images", [])
        if isinstance(inner_images, list) and len(inner_images) > 0:
            first_item = inner_images[0]

            if isinstance(first_item, str) and first_item.startswith("http"):
                return {
                    "image2url": first_item
                }

            if isinstance(first_item, dict):
                for key in ["url", "image_url", "file_url", "uri"]:
                    val = first_item.get(key)
                    if isinstance(val, str) and val.startswith("http"):
                        return {
                            "image2url": val
                        }

    # 抓不到就回空字串
    return {
        "image2url": ""
    }
```
