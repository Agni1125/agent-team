import json
import re


async def main(args):
    params = getattr(args, "params", None) or {}

    raw = (
        params.get("input")
        or params.get("output")
        or params.get("data")
        or params.get("body")
        or params.get("competition")
        or params
    )

    # ===== 1. 安全解析 =====
    def safe_parse(value, max_depth=20):
        current = value

        for _ in range(max_depth):
            if isinstance(current, str):
                text = current.strip()
                if not text:
                    return None

                try:
                    current = json.loads(text)
                    continue
                except Exception:
                    return current

            return current

        return current

    def collect_payloads(value):
        value = safe_parse(value)

        if value is None:
            return []

        if isinstance(value, dict) and (
            "product_results" in value
            or "organic_results" in value
            or "competitor_products" in value
        ):
            return [value]

        if isinstance(value, dict):
            records = []
            for key in [
                "output",
                "input",
                "body",
                "data",
                "result",
                "results",
                "items",
                "competition"
            ]:
                if key in value:
                    records.extend(collect_payloads(value.get(key)))
            return records

        if isinstance(value, list):
            records = []
            for item in value:
                records.extend(collect_payloads(item))
            return records

        return []

    payloads = collect_payloads(raw)

    # ===== 2. 工具函数 =====
    def clean_text(value):
        if value is None:
            return ""
        if isinstance(value, (int, float)):
            return value
        if not isinstance(value, str):
            return ""
        return value.strip()

    def clean_brand(value):
        text = clean_text(value)
        if not isinstance(text, str):
            return ""

        text = re.sub(r"^Visit the\s+", "", text, flags=re.I)
        text = re.sub(r"\s+Store$", "", text, flags=re.I)
        return text.strip()

    def is_valid_url(value):
        value = clean_text(value)
        return isinstance(value, str) and (
            value.startswith("http://") or value.startswith("https://")
        )

    def is_valid_image_url(value):
        value = clean_text(value)

        if not is_valid_url(value):
            return False

        lower = value.lower()

        invalid_tokens = [
            "youtube.com",
            "youtu.be",
            ".mp4",
            ".m3u8",
            ".html",
            "watch?"
        ]

        for token in invalid_tokens:
            if token in lower:
                return False

        return True

    def extract_asin_from_url(url):
        url = clean_text(url)
        if not isinstance(url, str):
            return ""

        patterns = [
            r"/dp/([A-Z0-9]{10})",
            r"/gp/product/([A-Z0-9]{10})",
            r"/product-reviews/([A-Z0-9]{10})",
            r"asin=([A-Z0-9]{10})",
            r"pd_rd_i=([A-Z0-9]{10})"
        ]

        for pattern in patterns:
            match = re.search(pattern, url)
            if match:
                return match.group(1)

        return ""

    def make_amazon_url(asin, amazon_domain="amazon.com"):
        asin = clean_text(asin)
        amazon_domain = clean_text(amazon_domain) or "amazon.com"

        if not isinstance(asin, str):
            return ""

        if not re.fullmatch(r"[A-Z0-9]{10}", asin):
            return ""

        return f"https://www.{amazon_domain}/dp/{asin}/"

    def get_asin(item):
        if not isinstance(item, dict):
            return ""

        return (
            clean_text(item.get("asin"))
            or extract_asin_from_url(item.get("link_clean"))
            or extract_asin_from_url(item.get("link"))
            or extract_asin_from_url(item.get("url"))
            or extract_asin_from_url(item.get("product_url"))
        )

    def get_product_url(item, amazon_domain="amazon.com"):
        if not isinstance(item, dict):
            return ""

        asin = get_asin(item)

        if asin:
            return make_amazon_url(asin, amazon_domain)

        for key in [
            "product_url",
            "productUrl",
            "detail_url",
            "detailUrl",
            "source_url",
            "sourceUrl",
            "amazon_url",
            "amazonUrl",
            "asin_url",
            "asinUrl",
            "product_link",
            "productLink",
            "link_clean",
            "link",
            "url"
        ]:
            value = clean_text(item.get(key))

            if not is_valid_url(value):
                continue

            if "serpapi.com" in value:
                continue

            asin_from_url = extract_asin_from_url(value)
            if asin_from_url:
                return make_amazon_url(asin_from_url, amazon_domain)

            return value

        return ""

    def get_first_image(item):
        if not isinstance(item, dict):
            return ""

        for key in [
            "image_url",
            "imageUrl",
            "thumbnail_url",
            "thumbnailUrl",
            "thumbnail",
            "original",
            "original_url",
            "originalUrl",
            "source_image_url",
            "sourceImageUrl",
            "image"
        ]:
            value = item.get(key)
            if is_valid_image_url(value):
                return value

        for key in [
            "image_urls",
            "imageUrls",
            "images",
            "thumbnails",
            "thumbnail_urls",
            "thumbnailUrls"
        ]:
            values = item.get(key)

            if isinstance(values, list):
                for value in values:
                    if isinstance(value, str) and is_valid_image_url(value):
                        return value

                    if isinstance(value, dict):
                        image = get_first_image(value)
                        if image:
                            return image

        return ""

    def get_all_images(item, limit=5):
        images = []

        first = get_first_image(item)
        if first:
            images.append(first)

        if not isinstance(item, dict):
            return images

        for key in [
            "thumbnails",
            "images",
            "image_urls",
            "imageUrls",
            "thumbnail_urls",
            "thumbnailUrls"
        ]:
            values = item.get(key)

            if not isinstance(values, list):
                continue

            for value in values:
                image = ""

                if isinstance(value, str) and is_valid_image_url(value):
                    image = value
                elif isinstance(value, dict):
                    image = get_first_image(value)

                if image and image not in images:
                    images.append(image)

                if len(images) >= limit:
                    return images

        return images

    def get_review_count(item):
        if not isinstance(item, dict):
            return ""

        for key in [
            "reviews",
            "review_count",
            "reviews_count",
            "ratings_total",
            "rating_count"
        ]:
            value = item.get(key)
            if value not in [None, ""]:
                return value

        return ""

    def limit_list(value, max_items):
        if isinstance(value, list):
            return value[:max_items]
        return []

    # ===== 3. 评论资料精简 =====
    def clean_review_data(data):
        reviews_info = data.get("reviews_information", {})
        summary = reviews_info.get("summary", {}) if isinstance(reviews_info, dict) else {}

        insights = []

        for insight in limit_list(summary.get("insights", []), 6):
            if not isinstance(insight, dict):
                continue

            insights.append({
                "topic": clean_text(insight.get("title")),
                "sentiment": clean_text(insight.get("sentiment")),
                "mentions": insight.get("mentions", {}),
                "summary": clean_text(insight.get("summary"))
            })

        return {
            "summary_text": clean_text(summary.get("text")),
            "insights": insights,
            "customer_reviews_distribution": summary.get("customer_reviews", {})
        }

    # ===== 4. 主商品清洗 =====
    def clean_product_payload(data, index):
        search_parameters = data.get("search_parameters", {})
        amazon_domain = search_parameters.get("amazon_domain", "amazon.com")

        product = data.get("product_results", {})
        product_details = data.get("product_details", {})
        item_specifications = data.get("item_specifications", {})

        if not isinstance(product, dict) or not product:
            return None

        asin = (
            get_asin(product)
            or clean_text(product_details.get("asin"))
            or clean_text(search_parameters.get("asin"))
        )

        if asin:
            product["asin"] = asin

        product_url = get_product_url(product, amazon_domain)
        image_urls = get_all_images(product, limit=5)

        about_item = data.get("about_item", [])
        if not isinstance(about_item, list):
            about_item = []

        main_claims = [
            clean_text(x)
            for x in about_item
            if clean_text(x)
        ][:5]

        return {
            "source_type": "amazon_product_detail",
            "position": index + 1,
            "asin": asin,
            "brand": clean_brand(product.get("brand")),
            "title": clean_text(product.get("title")),
            "product_url": product_url,
            "image_url": image_urls[0] if image_urls else "",
            "image_urls": image_urls,

            "price": product.get("price", ""),
            "extracted_price": product.get("extracted_price", ""),
            "price_unit": product.get("price_unit", ""),
            "rating": product.get("rating", ""),
            "reviews": get_review_count(product),
            "bought_last_month": product.get("bought_last_month", ""),
            "badges": product.get("badges", []),
            "discount": product.get("discount", ""),
            "stock": product.get("stock", ""),

            "main_claims": main_claims,

            "specifications": {
                "item_specifications": item_specifications if isinstance(item_specifications, dict) else {},
                "product_details": {
                    "scent": product_details.get("scent", ""),
                    "item_form": product_details.get("item_form", ""),
                    "scent_duration": product_details.get("scent_duration", ""),
                    "container_type": product_details.get("container_type", ""),
                    "brand_name": product_details.get("brand_name", ""),
                    "manufacturer": product_details.get("manufacturer", ""),
                    "model_number": product_details.get("model_number", ""),
                    "color": product_details.get("color", ""),
                    "product_dimensions": product_details.get("product_dimensions", ""),
                    "item_dimensions": product_details.get("item_dimensions", ""),
                    "date_first_available": product_details.get("date_first_available", ""),
                    "best_sellers_rank": product_details.get("best_sellers_rank", [])
                }
            },

            "review_data": clean_review_data(data)
        }

    # ===== 5. 参考商品清洗 =====
    def clean_reference_item(item, source_type, amazon_domain, source_asin=""):
        if not isinstance(item, dict):
            return None

        asin = get_asin(item)

        if source_asin and asin and asin == source_asin:
            return None

        title = clean_text(item.get("title")) or clean_text(item.get("name"))
        product_url = ""

        if asin:
            item["asin"] = asin
            product_url = make_amazon_url(asin, amazon_domain)
        else:
            product_url = get_product_url(item, amazon_domain)

        image_url = get_first_image(item)

        if not asin and not title and not product_url and not image_url:
            return None

        return {
            "source_type": source_type,
            "position": item.get("position", ""),
            "asin": asin,
            "brand": clean_brand(item.get("brand")),
            "title": title,
            "product_url": product_url,
            "image_url": image_url,

            "price": item.get("price", ""),
            "extracted_price": item.get("extracted_price", ""),
            "price_unit": item.get("price_unit", ""),
            "rating": item.get("rating", ""),
            "reviews": get_review_count(item),
            "badges": item.get("badges", []),
            "sponsored": item.get("sponsored", False),
            "stock": item.get("stock", "")
        }

    def extract_compare_with_similar(data):
        items = []
        product_description = data.get("product_description", [])

        if not isinstance(product_description, list):
            return items

        for block in product_description:
            if not isinstance(block, dict):
                continue

            compare_items = block.get("compare_with_similar", [])

            if isinstance(compare_items, list):
                items.extend(compare_items)

        return items

    # ===== 6. 批次清洗 =====
    competitor_products = []
    reference_products = []

    seen_main = set()
    seen_ref = set()

    for data in payloads:
        data = safe_parse(data)

        if not isinstance(data, dict):
            continue

        search_parameters = data.get("search_parameters", {})
        amazon_domain = search_parameters.get("amazon_domain", "amazon.com")

        # 已经是整理过的资料时，直接吃关键字段
        if "competitor_products" in data and "product_results" not in data:
            for item in data.get("competitor_products", []):
                if not isinstance(item, dict):
                    continue

                key = item.get("asin") or item.get("product_url") or item.get("title")

                if key and key not in seen_main:
                    seen_main.add(key)
                    competitor_products.append(item)

            for item in data.get("reference_products", []) or data.get("related_products", []):
                if not isinstance(item, dict):
                    continue

                key = item.get("asin") or item.get("product_url") or item.get("title")

                if key and key not in seen_ref:
                    seen_ref.add(key)
                    reference_products.append(item)

            continue

        # amazon_product 商品详情
        if "product_results" in data:
            cleaned_main = clean_product_payload(data, len(competitor_products))

            if cleaned_main:
                key = (
                    cleaned_main.get("asin")
                    or cleaned_main.get("product_url")
                    or cleaned_main.get("title")
                )

                if key and key not in seen_main:
                    seen_main.add(key)
                    competitor_products.append(cleaned_main)

                source_asin = cleaned_main.get("asin", "")

                source_groups = [
                    ("related_products", data.get("related_products", [])),
                    ("sponsored_brands", data.get("sponsored_brands", [])),
                    ("bought_together", data.get("bought_together", [])),
                    ("similar_products", data.get("similar_products", [])),
                    ("customers_also_bought", data.get("customers_also_bought", [])),
                    ("customers_also_viewed", data.get("customers_also_viewed", [])),
                    ("compare_with_similar", extract_compare_with_similar(data))
                ]

                for source_type, items in source_groups:
                    if not isinstance(items, list):
                        continue

                    for item in items:
                        cleaned_ref = clean_reference_item(
                            item=item,
                            source_type=source_type,
                            amazon_domain=amazon_domain,
                            source_asin=source_asin
                        )

                        if not cleaned_ref:
                            continue

                        key = (
                            cleaned_ref.get("asin")
                            or cleaned_ref.get("product_url")
                            or cleaned_ref.get("title")
                        )

                        if key and key not in seen_ref:
                            seen_ref.add(key)
                            reference_products.append(cleaned_ref)

        # amazon search 搜索页
        elif "organic_results" in data:
            organic_results = data.get("organic_results", [])

            if isinstance(organic_results, list):
                for item in organic_results:
                    cleaned_ref = clean_reference_item(
                        item=item,
                        source_type="organic_results",
                        amazon_domain=amazon_domain,
                        source_asin=""
                    )

                    if not cleaned_ref:
                        continue

                    key = (
                        cleaned_ref.get("asin")
                        or cleaned_ref.get("product_url")
                        or cleaned_ref.get("title")
                    )

                    if key and key not in seen_ref:
                        seen_ref.add(key)
                        reference_products.append(cleaned_ref)

    # ===== 7. 控制输出长度 =====
    competitor_products = competitor_products[:20]
    reference_products = reference_products[:40]

    asin_array = []

    for item in competitor_products + reference_products:
        asin = clean_text(item.get("asin"))
        if asin and asin not in asin_array:
            asin_array.append(asin)

    # ===== 8. 最终给报告节点的结构 =====
    result = {
        "data_type": "competitor_reference_data",
        "metadata": {
            "parsed_payload_count": len(payloads),
            "competitor_products_count": len(competitor_products),
            "reference_products_count": len(reference_products),
            "asin_array": asin_array
        },
        "data_status": {
            "status": "usable" if competitor_products or reference_products else "insufficient",
            "has_image_urls": any(
                item.get("image_url")
                for item in competitor_products + reference_products
            ),
            "has_product_urls": any(
                item.get("product_url")
                for item in competitor_products + reference_products
            )
        },
        "competitor_products": competitor_products,
        "reference_products": reference_products,
        "summary_for_report": {
            "主要競品數": len(competitor_products),
            "參考商品數": len(reference_products),
            "可顯示圖片數": sum(
                1 for item in competitor_products + reference_products
                if item.get("image_url")
            ),
            "可點擊商品頁數": sum(
                1 for item in competitor_products + reference_products
                if item.get("product_url")
            ),
            "可用ASIN": asin_array
        }
    }

    return {
        "competition": json.dumps(result, ensure_ascii=False)
    }
