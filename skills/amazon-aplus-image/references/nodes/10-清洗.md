```python
import json
import re
import html

async def main(args: Args) -> Output:
    params = args.params
    raw_input = params.get("input", "")

    def clean_text(value):
        if value is None:
            return ""
        value = str(value)
        value = html.unescape(value)
        value = value.replace("\\/", "/")
        value = value.replace("\\n", " ")
        value = value.replace("\\t", " ")
        value = re.sub(r"<[^>]*>", " ", value)
        value = re.sub(r"\s+", " ", value)
        return value.strip()

    def unwrap_text(value):
        """
        把 Coze 常見的多層 input / body / output 包裝拆掉。
        即使 JSON 不完整，也盡量還原文字。
        """
        if value is None:
            return ""

        current = value

        for _ in range(8):
            if isinstance(current, dict):
                for key in ["input", "body", "data", "output", "result"]:
                    if key in current:
                        current = current.get(key)
                        break
                else:
                    return json.dumps(current, ensure_ascii=False)

            elif isinstance(current, str):
                text = current.strip()

                # 先嘗試完整 JSON 解析
                try:
                    obj = json.loads(text)
                    current = obj
                    continue
                except Exception:
                    pass

                # 針對 {"input":"...."} 但整體 JSON 不完整的情況，直接抓 input 內文
                m = re.search(r'"input"\s*:\s*"(.*)', text, flags=re.S)
                if m:
                    text = m.group(1)

                    # 去掉尾部可能殘留的 "}
                    text = re.sub(r'"\s*\}\s*$', '', text, flags=re.S)

                # 反轉義
                text = text.replace('\\"', '"')
                text = text.replace("\\\\n", "\\n")
                text = text.replace("\\n", "\n")
                text = text.replace("\\\\t", "\\t")
                text = text.replace("\\t", " ")
                text = text.replace("\\/", "/")
                text = text.replace("\\\\", "\\")

                return text

            else:
                return str(current)

        return str(current)

    def try_parse_json(text):
        """
        嘗試把文字轉成 JSON。
        如果被截斷，會失敗，後面改用 regex 抓資料。
        """
        if not text:
            return {}

        text = text.strip()

        try:
            return json.loads(text)
        except Exception:
            pass

        first = text.find("{")
        last = text.rfind("}")

        if first != -1 and last != -1 and last > first:
            candidate = text[first:last + 1]
            try:
                return json.loads(candidate)
            except Exception:
                pass

        return {}

    def extract_by_regex(text):
        """
        當 JSON 被截斷或多層轉義導致不能 parse 時，
        直接從原文字串抓 Google 搜尋、影片、產品訊號。
        """
        result = {
            "query": "",
            "total_results": None,
            "organic_results_state": "",
            "inline_videos": [],
            "products": [],
            "local_places": []
        }

        # query
        q_match = re.search(r'"q"\s*:\s*"([^"]+)"', text)
        if not q_match:
            q_match = re.search(r'"query_displayed"\s*:\s*"([^"]+)"', text)

        if q_match:
            result["query"] = clean_text(q_match.group(1))

        # total_results
        total_match = re.search(r'"total_results"\s*:\s*(\d+)', text)
        if total_match:
            try:
                result["total_results"] = int(total_match.group(1))
            except Exception:
                result["total_results"] = None

        # organic state
        state_match = re.search(r'"organic_results_state"\s*:\s*"([^"]+)"', text)
        if state_match:
            result["organic_results_state"] = clean_text(state_match.group(1))

        # inline_videos 區塊
        video_area = ""
        video_start = text.find('"inline_videos"')
        product_start = text.find('"immersive_products"')

        if video_start != -1:
            if product_start != -1 and product_start > video_start:
                video_area = text[video_start:product_start]
            else:
                video_area = text[video_start:]

        if video_area:
            video_blocks = re.findall(r'\{(.*?)\}', video_area, flags=re.S)

            for block in video_blocks:
                title = re.search(r'"title"\s*:\s*"([^"]*)"', block)
                snippet = re.search(r'"snippet"\s*:\s*"([^"]*)"', block)
                channel = re.search(r'"channel"\s*:\s*"([^"]*)"', block)
                duration = re.search(r'"duration"\s*:\s*"([^"]*)"', block)
                date = re.search(r'"date"\s*:\s*"([^"]*)"', block)

                item = {
                    "title": clean_text(title.group(1)) if title else "",
                    "snippet": clean_text(snippet.group(1)) if snippet else "",
                    "channel": clean_text(channel.group(1)) if channel else "",
                    "duration": clean_text(duration.group(1)) if duration else "",
                    "date": clean_text(date.group(1)) if date else ""
                }

                if item["title"] or item["snippet"]:
                    result["inline_videos"].append(item)

        # products 區塊
        product_area = ""
        product_start = text.find('"immersive_products"')

        if product_start != -1:
            product_area = text[product_start:]

        if product_area:
            # 用 title/source/price/rating/reviews 抓，不依賴完整 JSON object
            product_titles = list(re.finditer(r'"title"\s*:\s*"([^"]+)"', product_area))

            for idx, title_match in enumerate(product_titles):
                start = title_match.start()
                end = product_titles[idx + 1].start() if idx + 1 < len(product_titles) else min(len(product_area), start + 3000)
                block = product_area[start:end]

                title = clean_text(title_match.group(1))

                source = re.search(r'"source"\s*:\s*"([^"]*)"', block)
                category = re.search(r'"category"\s*:\s*"([^"]*)"', block)
                price = re.search(r'"price"\s*:\s*"([^"]*)"', block)
                rating = re.search(r'"rating"\s*:\s*([0-9.]+)', block)
                reviews = re.search(r'"reviews"\s*:\s*([0-9]+)', block)

                # 避免把 image/source_logo 之類非產品 title 抓進來
                if not title:
                    continue

                item = {
                    "title": title,
                    "source": clean_text(source.group(1)) if source else "",
                    "category": clean_text(category.group(1)) if category else "",
                    "price": clean_text(price.group(1)) if price else "",
                    "rating": float(rating.group(1)) if rating else None,
                    "reviews": int(reviews.group(1)) if reviews else None
                }

                result["products"].append(item)

        # local places
        local_area = ""
        local_start = text.find('"local_results"')
        video_start = text.find('"inline_videos"')

        if local_start != -1:
            if video_start != -1 and video_start > local_start:
                local_area = text[local_start:video_start]
            else:
                local_area = text[local_start:]

        if local_area:
            place_titles = list(re.finditer(r'"title"\s*:\s*"([^"]+)"', local_area))

            for title_match in place_titles[:10]:
                title = clean_text(title_match.group(1))
                if title:
                    result["local_places"].append({
                        "title": title
                    })

        return result

    def extract_from_json(parsed):
        search_parameters = parsed.get("search_parameters", {}) or {}
        search_information = parsed.get("search_information", {}) or {}

        query = clean_text(
            search_parameters.get("q")
            or search_information.get("query_displayed")
            or ""
        )

        inline_videos = []
        for item in parsed.get("inline_videos", []) or []:
            if not isinstance(item, dict):
                continue

            inline_videos.append({
                "title": clean_text(item.get("title", "")),
                "snippet": clean_text(item.get("snippet", "")),
                "channel": clean_text(item.get("channel", "")),
                "duration": clean_text(item.get("duration", "")),
                "date": clean_text(item.get("date", ""))
            })

        products = []
        for key in ["immersive_products", "shopping_results"]:
            for item in parsed.get(key, []) or []:
                if not isinstance(item, dict):
                    continue

                products.append({
                    "title": clean_text(item.get("title", "")),
                    "source": clean_text(item.get("source", "")),
                    "category": clean_text(item.get("category", "")),
                    "price": clean_text(item.get("price", "")),
                    "rating": item.get("rating", None),
                    "reviews": item.get("reviews", None)
                })

        local_places = []
        local_results = parsed.get("local_results", {}) or {}
        for item in local_results.get("places", []) or []:
            if not isinstance(item, dict):
                continue

            local_places.append({
                "title": clean_text(item.get("title", "")),
                "type": clean_text(item.get("type", "")),
                "rating": item.get("rating", None),
                "reviews": item.get("reviews", None),
                "extensions": [clean_text(x) for x in item.get("extensions", []) if x]
            })

        return {
            "query": query,
            "total_results": search_information.get("total_results", None),
            "organic_results_state": clean_text(search_information.get("organic_results_state", "")),
            "inline_videos": inline_videos,
            "products": products,
            "local_places": local_places
        }

    try:
        text = unwrap_text(raw_input)
        parsed = try_parse_json(text)

        if isinstance(parsed, dict) and parsed:
            extracted = extract_from_json(parsed)
            parse_mode = "json"
        else:
            extracted = extract_by_regex(text)
            parse_mode = "regex_fallback"

        output = {
            "status": "success" if extracted.get("query") or extracted.get("products") or extracted.get("inline_videos") else "partial_success",
            "source_type": "google_search",
            "parse_mode": parse_mode,
            "query": extracted.get("query", ""),
            "search_information": {
                "total_results": extracted.get("total_results", None),
                "organic_results_state": extracted.get("organic_results_state", "")
            },
            "inline_videos": extracted.get("inline_videos", [])[:10],
            "products": extracted.get("products", [])[:20],
            "local_places": extracted.get("local_places", [])[:10],
            "quick_signal_summary": {
                "has_search_query": bool(extracted.get("query", "")),
                "has_inline_videos": len(extracted.get("inline_videos", [])) > 0,
                "has_products": len(extracted.get("products", [])) > 0,
                "has_local_places": len(extracted.get("local_places", [])) > 0,
                "product_count": len(extracted.get("products", [])),
                "video_count": len(extracted.get("inline_videos", []))
            },
            "data_warning": "If parse_mode is regex_fallback, the original input may be escaped, double-wrapped, or truncated. Long fields such as immersive_product_page_token should be removed upstream."
        }

        return {
            "output": json.dumps(output, ensure_ascii=False)
        }

    except Exception as e:
        error_output = {
            "status": "error",
            "message": str(e),
            "debug_preview": str(raw_input)[:500]
        }

        return {
            "output": json.dumps(error_output, ensure_ascii=False)
        }
```
