import json
import re
import html


async def main(args: Args) -> Output:
    params = args.params

    # 兼容不同上游變量名稱
    raw_input = (
        params.get("youtube_short")
        or params.get("input")
        or params.get("body")
        or params.get("data")
        or params
    )

    def clean_text(value):
        if value is None:
            return ""

        s = str(value)
        s = html.unescape(s)
        s = re.sub(r"<[^>]*>", " ", s)
        s = s.replace("\\/", "/")
        s = s.replace("\\n", " ").replace("\n", " ")
        s = s.replace("\\t", " ").replace("\t", " ")
        s = re.sub(r"\s+", " ", s)

        return s.strip()

    def deep_parse(value):
        """
        遞迴解析多層 JSON 字串。
        可處理：
        1. {"body": "{...serpapi json...}"}
        2. {"youtube_short": "{\"body\":\"...\"}"}
        3. markdown code block 包住的 JSON
        4. 已經是 dict 的資料
        """
        current = value

        for _ in range(15):
            # 如果已經是 SerpAPI 主結構，直接回傳
            if isinstance(current, dict) and (
                "shorts_results" in current
                or "shortsResults" in current
                or "inlineShortsResults" in current
                or "video_results" in current
            ):
                return current

            # 如果是 wrapper dict，往內拆
            if isinstance(current, dict):
                unwrapped = False

                for key in [
                    "youtube_short",
                    "body",
                    "data",
                    "result",
                    "output",
                    "processed_text",
                    "content",
                    "text"
                ]:
                    if key in current and current[key] not in [None, ""]:
                        current = current[key]
                        unwrapped = True
                        break

                if unwrapped:
                    continue

                return current

            # 如果是 list，不再解析
            if isinstance(current, list):
                return current

            # 如果是字串，嘗試轉 JSON
            if isinstance(current, str):
                s = current.strip()

                # 清除 markdown code block
                s = re.sub(r"^```(?:json)?\s*", "", s)
                s = re.sub(r"\s*```$", "", s)
                s = s.strip()

                try:
                    current = json.loads(s)
                    continue
                except Exception:
                    pass

                # 處理被多層 escape 的 JSON 字串
                try:
                    decoded = bytes(s, "utf-8").decode("unicode_escape")
                    current = json.loads(decoded)
                    continue
                except Exception:
                    return s

            return current

        return current

    def get_first_value(item, keys):
        if not isinstance(item, dict):
            return ""

        for key in keys:
            value = item.get(key)
            if value not in [None, ""]:
                return clean_text(value)

        return ""

    def extract_video_id(item):
        if not isinstance(item, dict):
            return ""

        # 兼容 SerpAPI snake_case / camelCase
        video_id = get_first_value(item, [
            "video_id",
            "videoId",
            "id"
        ])

        link = get_first_value(item, [
            "link",
            "url",
            "video_url",
            "videoUrl"
        ])

        # 從 Shorts URL 抽取
        if not video_id and "/shorts/" in link:
            video_id = link.split("/shorts/")[-1].split("?")[0].strip("/")

        # 從一般 YouTube watch URL 抽取
        if not video_id and "watch?v=" in link:
            video_id = link.split("watch?v=")[-1].split("&")[0].strip()

        # 從 youtu.be 短連結抽取
        if not video_id and "youtu.be/" in link:
            video_id = link.split("youtu.be/")[-1].split("?")[0].strip("/")

        return video_id.strip()

    def extract_views(item):
        if not isinstance(item, dict):
            return 0

        raw_views = (
            item.get("views")
            or item.get("viewCount")
            or item.get("views_original")
            or item.get("viewsOriginal")
            or 0
        )

        if isinstance(raw_views, int):
            return raw_views

        s = clean_text(raw_views).lower()
        s = s.replace(",", "")

        try:
            if "m" in s:
                return int(float(re.sub(r"[^0-9.]", "", s)) * 1000000)
            if "k" in s:
                return int(float(re.sub(r"[^0-9.]", "", s)) * 1000)

            num = re.sub(r"[^0-9]", "", s)
            return int(num) if num else 0

        except Exception:
            return 0

    def extract_video_ids(parsed):
        videos = {}

        def upsert_video_id(item):
            video_id = extract_video_id(item)

            if not video_id:
                return

            views = extract_views(item)

            # 去重：同一個 video_id 只保留最高觀看數
            if video_id not in videos:
                videos[video_id] = views
            else:
                videos[video_id] = max(videos[video_id], views)

        # 情況 1：SerpAPI 格式 shorts_results
        if isinstance(parsed, dict):
            for group in parsed.get("shorts_results", []) or []:
                if not isinstance(group, dict):
                    continue

                for item in group.get("shorts", []) or []:
                    upsert_video_id(item)

        # 情況 2：camelCase 格式 shortsResults
        if isinstance(parsed, dict):
            for group in parsed.get("shortsResults", []) or []:
                if not isinstance(group, dict):
                    continue

                for item in group.get("shorts", []) or []:
                    upsert_video_id(item)

        # 情況 3：inlineShortsResults
        if isinstance(parsed, dict):
            for item in parsed.get("inlineShortsResults", []) or []:
                upsert_video_id(item)

        # 情況 4：如果上游直接給 array
        if isinstance(parsed, list):
            for item in parsed:
                upsert_video_id(item)

        # 依觀看數高到低排序，只輸出 video_id array
        video_ids = sorted(
            videos.keys(),
            key=lambda vid: videos.get(vid, 0),
            reverse=True
        )

        return video_ids

    try:
        parsed = deep_parse(raw_input)
        video_ids = extract_video_ids(parsed)

        return {
            "YT_output": video_ids
        }

    except Exception as e:
        return {
            "YT_output": [],
            "error": str(e)
        }
````
