import json
import re
import html


async def main(args: Args) -> Output:
    params = args.params
    raw_input = params.get("input", "")

    def decode_json_string(s):
        """
        把 Reddit JSON 字符串里的转义符号还原：
        \\n -> 换行
        \\\" -> "
        \\u0026amp; -> &
        """
        if s is None:
            return ""

        s = str(s)

        # 先尝试用 JSON 字符串方式解码
        try:
            s = json.loads('"' + s.replace('"', '\\"') + '"')
        except Exception:
            try:
                s = bytes(s, "utf-8").decode("unicode_escape")
            except Exception:
                pass

        # HTML entity 反转义
        s = html.unescape(s)

        # 清掉 HTML 标签
        s = re.sub(r"<[^>]*>", " ", s)

        # 常见残留处理
        s = s.replace("\\u0026amp;", "&")
        s = s.replace("\\u0026lt;", "<")
        s = s.replace("\\u0026gt;", ">")
        s = s.replace("\\u0026quot;", '"')
        s = s.replace("\\/", "/")

        # 把字面上的 \n 变成真正换行
        s = s.replace("\\n", "\n")
        s = s.replace("\\t", " ")

        # 把过多空格清掉，但保留段落换行
        s = re.sub(r"[ \t]+", " ", s)
        s = re.sub(r"\n{3,}", "\n\n", s)

        return s.strip()

    def parse_deep(value):
        """
        尝试解析完整 JSON。
        如果输入是 {"body":"{...}"}，会继续解析 body。
        """
        current = value

        for _ in range(8):
            if not isinstance(current, str):
                break

            try:
                current = json.loads(current)
            except Exception:
                break

        for _ in range(8):
            if isinstance(current, dict) and isinstance(current.get("body"), str):
                current = parse_deep(current.get("body"))
                continue

            if isinstance(current, dict) and isinstance(current.get("Reddit_comment"), str):
                current = parse_deep(current.get("Reddit_comment"))
                continue

            if isinstance(current, dict) and isinstance(current.get("data"), str):
                current = parse_deep(current.get("data"))
                continue

            break

        return current

    def extract_from_json(parsed):
        """
        如果能解析成 JSON，就从 posts[].data.selftext 取正文。
        """
        posts = []

        if isinstance(parsed, dict):
            data = parsed.get("data")

            if isinstance(data, dict) and isinstance(data.get("posts"), list):
                posts = data.get("posts")

            elif isinstance(parsed.get("posts"), list):
                posts = parsed.get("posts")

        results = []

        for item in posts:
            if not isinstance(item, dict):
                continue

            d = item.get("data", item)
            if not isinstance(d, dict):
                continue

            title = decode_json_string(d.get("title", ""))
            selftext = decode_json_string(d.get("selftext", ""))

            if not selftext:
                continue

            block = ""

            if title:
                block += f"Title: {title}\n\n"

            block += selftext
            results.append(block)

        return results

    def extract_from_raw_text(text):
        """
        如果 JSON 解析失败，就直接从原始字符串里用 regex 抓 selftext。
        重点：抓 selftext 到 selftext_html 之前，保留长段正文。
        """
        if not text:
            return []

        text = str(text)

        patterns = [
            # 处理 \"selftext\":\"....\",\"selftext_html\"
            r'\\"selftext\\"\s*:\s*\\"(.*?)\\"\s*,\s*\\"selftext_html\\"',

            # 处理 "selftext":"....","selftext_html"
            r'"selftext"\s*:\s*"(.*?)"\s*,\s*"selftext_html"',

            # 处理 selftext\":\"....\",\"selftext_html
            r'selftext\\?":\\?"(.*?)\\?",\\?"selftext_html',

            # 只抓 selftext 后面内容，作为兜底
            r'selftext\\?":\\?"(.*)'
        ]

        results = []

        for pattern in patterns:
            matches = re.findall(pattern, text, flags=re.S)

            for m in matches:
                decoded = decode_json_string(m)

                # 避免抓到空值或 HTML 版内容
                if not decoded:
                    continue

                if "SC_OFF" in decoded or "<div" in decoded or "&lt;div" in decoded:
                    continue

                # 太短的不要
                if len(decoded) < 80:
                    continue

                results.append(decoded)

            if results:
                break

        return results

    try:
        parsed = parse_deep(raw_input)

        # 优先从 JSON 结构取 selftext
        extracted_texts = extract_from_json(parsed)

        # 如果 JSON 解析不到，再从原始文本 regex 抓 selftext
        if not extracted_texts:
            extracted_texts = extract_from_raw_text(raw_input)

        if not extracted_texts:
            return {
                "output": "No selftext extracted. 请确认 input 接的是 HTTP body 原始输出，不要接 text_processor 截断后的 processed_text。"
            }

        # 保留长段文字，但避免无限爆 token
        # 你可以把 30000 调大或调小
        final_text = "\n\n---\n\n".join(extracted_texts)
        final_text = final_text[:30000]

        return {
            "output": final_text
        }

    except Exception as e:
        return {
            "output": f"error={str(e)}"
        }
