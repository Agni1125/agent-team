// 在这里，您可以通过 'params'  获取节点中的输入变量，并通过 'ret' 输出结果
// 'params' 已经被正确地注入到环境中
async function main({ params }: Args): Promise<Output> {
    // 1. 获取输入的Input数组，兜底空数组防止入参非数组/无值时报错
    const inputImages = params.input || [];
    // 2. 提取数组中第一个、第二个图片元素（索引0=第一个，索引1=第二个）
    const firstImage = inputImages[0]; // 第一张图片
    const secondImage = inputImages[2]; // 第三张图片

    // 3. 构建输出对象，key可根据需求修改（比如img1/img2，保持语义清晰即可）
    const ret = {
        "firstImage": firstImage,
        "secondImage": secondImage
    };

    return ret;
}
