```python
import json
import re


async def main(args):
    # ===== 1. 读取输入参数 =====
    params = getattr(args, "params", None) or {}

    # 你的输入变量名是 imageurl
    raw = (
        params.get("imageurl")
        or params.get("image_url")
        or params.get("image_urls")
        or params.get("images")
        or params.get("output")
        or params.get("data")
        or params.get("result")
        or params.get("input")
        or params
    )

    # ===== 2. 安全解析 JSON 字符串 =====
    def safe_parse(value, max_depth=5):
        current = value

        for _ in range(max_depth):
            if not isinstance(current, str):
                return current

            text = current.strip()

            if not (
                (text.startswith("{") and text.endswith("}"))
                or (text.startswith("[") and text.endswith("]"))
            ):
                return current

            try:
                current = json.loads(text)
            except Exception:
                return current

        return current

    # ===== 3. 提取 URL =====
    image_urls = []

    def add_url(url):
        if not isinstance(url, str):
            return

        url = url.strip()

        if not re.match(r"^https?://", url, re.I):
            return

        url = url.replace("&amp;", "&").rstrip("，。,.);]}\"'")

        if url not in image_urls:
            image_urls.append(url)

    def extract_from_text(text):
        if not isinstance(text, str):
            return

        matches = re.findall(r"https?://[^\s\"'<>\\]+", text)

        for url in matches:
            add_url(url)

    def walk(value):
        value = safe_parse(value)

        if value is None:
            return

        if isinstance(value, str):
            extract_from_text(value)
            return

        if isinstance(value, list):
            for item in value:
                walk(item)
            return

        if isinstance(value, dict):
            # 常见图片字段
            possible_keys = [
                "url",
                "uri",
                "src",
                "image",
                "image_url",
                "imageUrl",
                "image_ref",
                "imageRef",
                "origin_url",
                "preview_url",
                "download_url",
                "final_image_url",
                "output_url",
                "result_url",
            ]

            for key in possible_keys:
                if key in value:
                    walk(value.get(key))

            # 兜底扫描所有字段
            for key, item in value.items():
                if key not in possible_keys:
                    walk(item)

            return

    walk(raw)

    # ===== 4. 输出 String =====
    # 如果下游要的是「URL array 的字符串」
    final_images_text = json.dumps(image_urls, ensure_ascii=False)

    return {
        "final_images_text": final_images_text
    }
```
