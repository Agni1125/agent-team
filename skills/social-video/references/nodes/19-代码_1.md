```python
import json
import re


async def main(args: Args) -> Output:
    params = args.params

    raw = params.get("input", "")

    # 如果 input 沒拿到，就嘗試拿整包 params
    if raw in [None, ""]:
        raw = params

    def extract_task_id(value):
        # 1. 如果是 dict，優先從 body 或 task_id 取
        if isinstance(value, dict):
            if value.get("task_id"):
                return str(value.get("task_id"))

            if value.get("body"):
                return extract_task_id(value.get("body"))

            # dict 轉字串後再用正則兜底
            text = json.dumps(value, ensure_ascii=False)
            return extract_task_id(text)

        # 2. 如果是字串，先嘗試多層 JSON parse
        if isinstance(value, str):
            text = value.strip()

            # 最多解析 5 層，處理 "{\"task_id\":\"xxx\"}" 這種情況
            current = text
            for _ in range(5):
                try:
                    parsed = json.loads(current)

                    if isinstance(parsed, dict):
                        if parsed.get("task_id"):
                            return str(parsed.get("task_id"))

                        if parsed.get("body"):
                            return extract_task_id(parsed.get("body"))

                        current = json.dumps(parsed, ensure_ascii=False)

                    elif isinstance(parsed, str):
                        current = parsed.strip()

                    else:
                        break

                except Exception:
                    break

            # 3. JSON parse 失敗時，用正則直接抓 task_id
            patterns = [
                r'"task_id"\s*:\s*"([^"]+)"',
                r'\\"task_id\\"\s*:\s*\\"([^\\"]+)\\"',
                r'task_id["\\]*\s*[:=]\s*["\\]*([0-9a-zA-Z\-]+)'
            ]

            for pattern in patterns:
                match = re.search(pattern, text)
                if match:
                    return match.group(1)

        return ""

    task_id = extract_task_id(raw)

    return {
        "viedo_ID": task_id
    }
```
