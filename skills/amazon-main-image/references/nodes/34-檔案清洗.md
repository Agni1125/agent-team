```python
import json
import re
import html

async def main(args):
    params = getattr(args, "params", None) or {}

    raw_input = (
        params.get("input")
        or params.get("outputList")
        or params.get("data")
        or params
    )

    # ===== 可調整區 =====
    MAX_BSR_ITEMS = 100
    MAX_REVIEWS = 500
    MAX_TOTAL_CHARS = 35000

    REVIEW_COLUMN = "内容"  # 要中文就改成：内容(翻译)

    def to_text(value):
        if isinstance(value, str):
            return value
        try:
            return json.dumps(value, ensure_ascii=False)
        except Exception:
            return str(value)

    def safe_parse(value, max_depth=10):
        current = value

        for _ in range(max_depth):
            if not isinstance(current, str):
                return current

            text = current.strip()

            try:
                current = json.loads(text)
                continue
            except Exception:
                return current

        return current

    def clean_cell(text):
        if text is None:
            return ""

        text = to_text(text)
        text = html.unescape(text)

        text = text.replace("\\n", "\n")
        text = text.replace("\\\"", "\"")
        text = text.replace("\\/", "/")

        text = re.sub(r"<br\s*/?>", "\n", text, flags=re.I)
        text = re.sub(r"<[^>]+>", " ", text)

        text = re.sub(r"[ \t]+", " ", text)
        text = re.sub(r"\n{3,}", "\n\n", text)
        return text.strip()

    def split_markdown_row(row):
        row = row.strip()

        if not row.startswith("|"):
            return []

        if row.endswith("|"):
            row = row[1:-1]
        else:
            row = row[1:]

        return [cell.strip() for cell in row.split("|")]

    def extract_data_blocks(obj):
        """
        從 read.outputList 裡抓每一份報表的 data 欄位。
        """
        obj = safe_parse(obj)
        blocks = []

        def walk(x):
            x = safe_parse(x)

            if isinstance(x, dict):
                data_value = x.get("data")

                if isinstance(data_value, str) and "|" in data_value:
                    blocks.append(data_value)

                for v in x.values():
                    if isinstance(v, (dict, list, str)):
                        walk(v)

            elif isinstance(x, list):
                for item in x:
                    walk(item)

            elif isinstance(x, str):
                if "|" in x:
                    blocks.append(x)

        walk(obj)
        return blocks

    def table_to_rows(table_text):
        table_text = clean_cell(table_text)
        lines = table_text.splitlines()

        header = None
        rows = []

        for line in lines:
            line = line.strip()

            if not line.startswith("|"):
                continue

            cells = split_markdown_row(line)

            if not cells:
                continue

            # 跳過 |---|---|
            if all(re.fullmatch(r"-+", c.replace(" ", "")) for c in cells if c):
                continue

            if header is None:
                header = cells
                continue

            row_obj = {}
            for i, col in enumerate(header):
                row_obj[col] = clean_cell(cells[i]) if i < len(cells) else ""

            rows.append(row_obj)

        return header or [], rows

    data_blocks = extract_data_blocks(raw_input)

    cleaned_bsr = []
    cleaned_reviews = []

    seen_bsr = set()
    seen_review = set()

    for block in data_blocks:
        header, rows = table_to_rows(block)

        header_text = "|".join(header)

        # ===== BSR 報表：只保留商品标题 / 产品卖点 =====
        if "商品标题" in header_text and "产品卖点" in header_text:
            for row in rows:
                title = clean_cell(row.get("商品标题"))
                selling_points = clean_cell(row.get("产品卖点"))

                if not title and not selling_points:
                    continue

                key = title + selling_points
                if key in seen_bsr:
                    continue

                seen_bsr.add(key)

                cleaned_bsr.append({
                    "商品标题": title,
                    "产品卖点": selling_points
                })

                if len(cleaned_bsr) >= MAX_BSR_ITEMS:
                    break

        # ===== Review 報表：只保留内容 =====
        elif "内容" in header_text and "星级" in header_text:
            for row in rows:
                review_text = clean_cell(row.get(REVIEW_COLUMN))

                if not review_text:
                    continue

                if review_text in seen_review:
                    continue

                seen_review.add(review_text)
                cleaned_reviews.append(review_text)

                if len(cleaned_reviews) >= MAX_REVIEWS:
                    break

    output_obj = {
        "bsr_competitor_data": cleaned_bsr,
        "review_data": cleaned_reviews,
        "data_cleaning_note": {
            "bsr_kept_fields": ["商品标题", "产品卖点"],
            "review_kept_field": REVIEW_COLUMN,
            "bsr_count": len(cleaned_bsr),
            "review_count": len(cleaned_reviews)
        }
    }

    clean_report = json.dumps(output_obj, ensure_ascii=False, indent=2)

    if len(clean_report) > MAX_TOTAL_CHARS:
        clean_report = clean_report[:MAX_TOTAL_CHARS] + "\n...[已截斷，避免超過下游模型長度限制]"

    return {
        "clean_report": clean_report
    }
```
