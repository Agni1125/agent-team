import json
import re
import html
from typing import Any


async def main(args: Args) -> Output:
    params = args.params

    raw_input = (
        params.get("output")
        or params.get("input")
        or params.get("batch_output")
        or params.get("data")
        or params
    )

    def clean_text(value: Any) -> str:
        if value is None:
            return ""

        s = str(value)
        s = html.unescape(s)
        s = s.replace("\\n", " ").replace("\n", " ")
        s = s.replace("\\r", " ").replace("\r", " ")
        s = s.replace("\\t", " ").replace("\t", " ")
        s = re.sub(r"\s+", " ", s)

        return s.strip()

    def parse_json_like(value: Any) -> Any:
        current = value

        for _ in range(8):
            if isinstance(current, (dict, list)):
                return current

            if not isinstance(current, str):
                return current

            s = current.strip()

            # 移除 markdown code block
            s = re.sub(r"^```(?:json)?\s*", "", s)
            s = re.sub(r"\s*```$", "", s)

            try:
                current = json.loads(s)
                continue
            except Exception:
                try:
                    decoded = bytes(s, "utf-8").decode("unicode_escape")
                    current = json.loads(decoded)
                    continue
                except Exception:
                    return value

        return current

    def normalize_batch_items(value: Any) -> list:
        parsed = parse_json_like(value)

        if isinstance(parsed, dict):
            for key in ["input", "output", "result", "data", "batch_output", "items"]:
                if key in parsed:
                    inner = parsed.get(key)
                    if isinstance(inner, list):
                        return inner
                    return [inner]

            if "video_id" in parsed or "comments" in parsed:
                return [parsed]

            return []

        if isinstance(parsed, list):
            return parsed

        if isinstance(parsed, str):
            return [parsed]

        return []

    def extract_comments_from_item(item: Any) -> list:
        parsed = parse_json_like(item)

        if not isinstance(parsed, dict):
            return []

        comments_raw = parsed.get("comments", []) or []

        if not isinstance(comments_raw, list):
            return []

        clean_comments = []

        for comment in comments_raw:
            if not isinstance(comment, dict):
                continue

            text = clean_text(comment.get("text", ""))
            published_time = clean_text(comment.get("published_time", ""))

            if not text:
                continue

            clean_comments.append({
                "published_time": published_time,
                "comment": text
            })

        return clean_comments

    try:
        batch_items = normalize_batch_items(raw_input)

        all_comments = []
        seen_texts = set()

        for item in batch_items:
            comments = extract_comments_from_item(item)

            for comment in comments:
                text = comment.get("comment", "")

                if not text:
                    continue

                # 全部影片留言去重
                if text in seen_texts:
                    continue

                seen_texts.add(text)
                all_comments.append(comment)

        return {
            "YT_comments_output": all_comments
        }

    except Exception:
        return {
            "YT_comments_output": []
        }
