import json
import re
import html

async def main(args):
    params = getattr(args, "params", None) or {}

    raw = (
        params.get("input")
        or params.get("raw_data")
        or params.get("competition")
        or params.get("reviews")
        or params.get("data")
        or params.get("body")
        or params
    )

    def to_text(value):
        if isinstance(value, str):
            return value
        try:
            return json.dumps(value, ensure_ascii=False)
        except Exception:
            return str(value)

    def clean_text(text):
        if not isinstance(text, str):
            return ""

        text = html.unescape(text)
        text = text.replace("\\n", " ")
        text = text.replace("\n", " ")
        text = text.replace('\\"', '"')
        text = text.replace("\\/", "/")
        text = re.sub(r"\s+", " ", text).strip()
        return text

    def normalize_raw_text(text):
        text = to_text(text)

        # 多跑幾輪，把 Coze / HTTP 包裝造成的跳脫符號拆掉
        for _ in range(10):
            old = text

            # 嘗試 JSON 解析
            try:
                parsed = json.loads(text)

                if isinstance(parsed, dict):
                    # 如果外層是 {"body": "..."}，直接拿 body
                    if "body" in parsed:
                        text = to_text(parsed.get("body"))
                    else:
                        text = json.dumps(parsed, ensure_ascii=False)

                elif isinstance(parsed, list):
                    text = json.dumps(parsed, ensure_ascii=False)

                elif isinstance(parsed, str):
                    text = parsed

            except Exception:
                pass

            # 手動解除常見跳脫
            text = text.replace('\\\\\\"', '"')
            text = text.replace('\\"', '"')
            text = text.replace("\\n", " ")
            text = text.replace("\\/", "/")

            if text == old:
                break

        return text

    raw_text = normalize_raw_text(raw)

    review_contents = []
    seen = set()

    def add_review(text):
        text = clean_text(text)
        if text and text not in seen:
            seen.add(text)
            review_contents.append(text)

    # 只從 authors_reviews 開始抓，避免抓到商品描述、summary、snippet
    start_positions = [
        raw_text.find('"authors_reviews"'),
        raw_text.find("authors_reviews")
    ]

    valid_positions = [pos for pos in start_positions if pos != -1]

    if valid_positions:
        start = min(valid_positions)
        review_section = raw_text[start:]

        # 抓 authors_reviews / other_countries_reviews 裡面的 text
        pattern = r'"text"\s*:\s*"((?:\\.|[^"\\])*)"'

        for match in re.finditer(pattern, review_section):
            text_value = match.group(1)

            # 還原 JSON 字串內容
            try:
                text_value = json.loads('"' + text_value + '"')
            except Exception:
                pass

            add_review(text_value)

    amazon_comment = "\n\n".join(review_contents)

    return {
        "amazon_comment": amazon_comment
    }
