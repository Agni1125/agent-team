import json
import re
import html


async def main(args: Args) -> Output:
    params = args.params
    raw_input = params.get("input", "")

    # =========================
    # 1. 基础清理
    # =========================
    def clean_text(value):
        if value is None:
            return ""

        text = str(value)

        # HTML entity 反转义
        text = html.unescape(text)

        # 移除 HTML 标签
        text = re.sub(r"<[^>]*>", " ", text)

        # 常见转义处理
        text = text.replace("\\u0026amp;", "&")
        text = text.replace("\\u0026lt;", "<")
        text = text.replace("\\u0026gt;", ">")
        text = text.replace("\\u0026quot;", '"')
        text = text.replace("\\/", "/")
        text = text.replace("\\n", "\n")
        text = text.replace("\\t", " ")

        # 清理空白
        text = re.sub(r"[ \t]+", " ", text)
        text = re.sub(r"\n{3,}", "\n\n", text)

        return text.strip()

    # =========================
    # 2. 深度解析 JSON
    # 兼容：
    # - HTTP body 原始 JSON 字符串
    # - {"body":"{...}"}
    # - {"input":"{...}"}
    # - {"output":"{...}"}
    # - OpenAI / Gemini choices.message.content
    # =========================
    def try_json_loads(value):
        if not isinstance(value, str):
            return value

        text = value.strip()

        # 去掉 markdown code block
        text = re.sub(r"^```(?:json)?", "", text).strip()
        text = re.sub(r"```$", "", text).strip()

        try:
            return json.loads(text)
        except Exception:
            pass

        # 有些平台会把 JSON 又包成字符串
        try:
            return json.loads(html.unescape(text))
        except Exception:
            pass

        return value

    def parse_deep(value, max_depth=10):
        current = value

        for _ in range(max_depth):
            before = current
            current = try_json_loads(current)

            # OpenAI / Gemini 标准返回结构
            if isinstance(current, dict):
                choices = current.get("choices")
                if isinstance(choices, list) and choices:
                    msg = choices[0].get("message", {})
                    content = msg.get("content")
                    if isinstance(content, str):
                        current = content
                        continue

            # Coze / HTTP 常见包裹字段
            if isinstance(current, dict):
                unwrapped = False

                for key in ["body", "input", "output", "result", "content"]:
                    if key in current and isinstance(current.get(key), str):
                        current = current.get(key)
                        unwrapped = True
                        break

                if unwrapped:
                    continue

            if current == before:
                break

        return current

    parsed = parse_deep(raw_input)

    # =========================
    # 3. 通用取值工具
    # =========================
    def pick(obj, *keys):
        if not isinstance(obj, dict):
            return ""

        for key in keys:
            value = obj.get(key)
            if value not in [None, "", [], {}]:
                return value

        return ""

    def as_list(value):
        if isinstance(value, list):
            return value
        return []

    def line_value(label, value):
        value = clean_text(value)
        if value:
            return f"- {label}: {value}"
        return ""

    def get_link(item):
        return pick(
            item,
            "link",
            "url",
            "source_url",
            "product_link",
            "serpapi_link"
        )

    def truncate(text, limit=500):
        text = clean_text(text)
        if len(text) > limit:
            return text[:limit].rstrip() + "..."
        return text

    # =========================
    # 4. 如果不是 JSON，就直接清理原文
    # =========================
    if not isinstance(parsed, (dict, list)):
        cleaned = clean_text(parsed)

        if not cleaned:
            return {
                "output": "No valid text extracted. 请确认 input 接的是 Google / SerpAPI HTTP body 原始输出。"
            }

        return {
            "output": cleaned[:30000]
        }

    # 如果最外层是 list，尝试包成 dict 处理
    if isinstance(parsed, list):
        parsed = {
            "items": parsed
        }

    # =========================
    # 5. SerpAPI / Google 搜寻结果提取
    # =========================
    lines = []
    lines.append("# Google Search / SerpAPI 清洗结果")

    # ---- 搜寻基础资讯 ----
    search_parameters = pick(parsed, "search_parameters", "parameters")
    search_metadata = pick(parsed, "search_metadata", "metadata")
    search_information = pick(parsed, "search_information", "search_info")

    if isinstance(search_parameters, dict):
        q = pick(search_parameters, "q", "query", "keyword")
        hl = pick(search_parameters, "hl")
        gl = pick(search_parameters, "gl")
        location = pick(search_parameters, "location", "uule")

        lines.append("\n## 搜寻基础资讯")
        if q:
            lines.append(f"- Keyword: {clean_text(q)}")
        if hl:
            lines.append(f"- Language: {clean_text(hl)}")
        if gl:
            lines.append(f"- Market: {clean_text(gl)}")
        if location:
            lines.append(f"- Location: {clean_text(location)}")

    if isinstance(search_information, dict):
        total_results = pick(search_information, "total_results")
        time_taken = pick(search_information, "time_taken_displayed", "time_taken")
        detected_location = pick(search_information, "detected_location")

        info_lines = []
        if total_results:
            info_lines.append(f"- Total results: {clean_text(total_results)}")
        if time_taken:
            info_lines.append(f"- Time taken: {clean_text(time_taken)}")
        if detected_location:
            info_lines.append(f"- Detected location: {clean_text(detected_location)}")

        if info_lines:
            lines.append("\n## 搜寻结果概况")
            lines.extend(info_lines)

    if isinstance(search_metadata, dict):
        status = pick(search_metadata, "status")
        google_url = pick(search_metadata, "google_url")

        meta_lines = []
        if status:
            meta_lines.append(f"- Status: {clean_text(status)}")
        if google_url:
            meta_lines.append(f"- Google URL: {clean_text(google_url)}")

        if meta_lines:
            lines.append("\n## 搜寻来源")
            lines.extend(meta_lines)

    # ---- Organic Results ----
    organic_results = as_list(pick(parsed, "organic_results"))
    if organic_results:
        lines.append("\n## 自然搜寻结果 Organic Results")

        for idx, item in enumerate(organic_results[:10], 1):
            if not isinstance(item, dict):
                continue

            title = pick(item, "title")
            source = pick(item, "source", "displayed_link")
            link = get_link(item)
            snippet = pick(item, "snippet", "description")

            lines.append(f"\n### Organic Result {idx}")
            if title:
                lines.append(f"- Title: {truncate(title, 180)}")
            if source:
                lines.append(f"- Source: {truncate(source, 120)}")
            if link:
                lines.append(f"- Link: {clean_text(link)}")
            if snippet:
                lines.append(f"- Snippet: {truncate(snippet, 500)}")

    # ---- Related Questions ----
    related_questions = as_list(pick(parsed, "related_questions"))
    if related_questions:
        lines.append("\n## Google 相关问题 Related Questions")

        for idx, item in enumerate(related_questions[:10], 1):
            if not isinstance(item, dict):
                continue

            question = pick(item, "question")
            snippet = pick(item, "snippet", "answer")
            title = pick(item, "title")
            link = get_link(item)

            lines.append(f"\n### Question {idx}")
            if question:
                lines.append(f"- Question: {truncate(question, 220)}")
            if snippet:
                lines.append(f"- Answer / Snippet: {truncate(snippet, 500)}")
            if title:
                lines.append(f"- Source title: {truncate(title, 180)}")
            if link:
                lines.append(f"- Link: {clean_text(link)}")

    # ---- Related Searches ----
    related_searches = as_list(pick(parsed, "related_searches"))
    if related_searches:
        lines.append("\n## 相关搜寻 Related Searches")

        for idx, item in enumerate(related_searches[:20], 1):
            if isinstance(item, dict):
                query = pick(item, "query", "text")
                link = get_link(item)
            else:
                query = item
                link = ""

            if query:
                line = f"- {idx}. {clean_text(query)}"
                if link:
                    line += f" | {clean_text(link)}"
                lines.append(line)

    # ---- Inline Images / Images Results ----
    image_results = []
    for key in ["inline_images", "images_results", "image_results"]:
        arr = as_list(pick(parsed, key))
        if arr:
            image_results = arr
            break

    if image_results:
        lines.append("\n## 图片结果 / 视觉参考 Images")

        for idx, item in enumerate(image_results[:12], 1):
            if not isinstance(item, dict):
                continue

            title = pick(item, "title")
            source = pick(item, "source")
            link = get_link(item)
            thumbnail = pick(item, "thumbnail", "thumbnail_url")
            original = pick(item, "original", "image", "image_url")

            lines.append(f"\n### Image Result {idx}")
            if title:
                lines.append(f"- Title: {truncate(title, 180)}")
            if source:
                lines.append(f"- Source: {truncate(source, 120)}")
            if link:
                lines.append(f"- Page link: {clean_text(link)}")
            if thumbnail:
                lines.append(f"- Thumbnail: {clean_text(thumbnail)}")
            if original:
                lines.append(f"- Image URL: {clean_text(original)}")

    # ---- Videos ----
    video_results = []
    for key in ["inline_videos", "short_videos", "video_results", "videos_results"]:
        arr = as_list(pick(parsed, key))
        if arr:
            video_results.extend(arr)

    if video_results:
        lines.append("\n## 视频结果 Videos")

        for idx, item in enumerate(video_results[:10], 1):
            if not isinstance(item, dict):
                continue

            title = pick(item, "title")
            source = pick(item, "source", "channel")
            link = get_link(item)
            snippet = pick(item, "snippet", "description")

            lines.append(f"\n### Video Result {idx}")
            if title:
                lines.append(f"- Title: {truncate(title, 180)}")
            if source:
                lines.append(f"- Source: {truncate(source, 120)}")
            if link:
                lines.append(f"- Link: {clean_text(link)}")
            if snippet:
                lines.append(f"- Snippet: {truncate(snippet, 400)}")

    # ---- Shopping Results ----
    shopping_results = as_list(pick(parsed, "shopping_results"))
    if shopping_results:
        lines.append("\n## 购物结果 Shopping Results")

        for idx, item in enumerate(shopping_results[:10], 1):
            if not isinstance(item, dict):
                continue

            title = pick(item, "title")
            source = pick(item, "source")
            price = pick(item, "price")
            rating = pick(item, "rating")
            reviews = pick(item, "reviews")
            link = get_link(item)

            lines.append(f"\n### Shopping Result {idx}")
            if title:
                lines.append(f"- Title: {truncate(title, 180)}")
            if source:
                lines.append(f"- Source: {truncate(source, 120)}")
            if price:
                lines.append(f"- Price: {clean_text(price)}")
            if rating:
                lines.append(f"- Rating: {clean_text(rating)}")
            if reviews:
                lines.append(f"- Reviews: {clean_text(reviews)}")
            if link:
                lines.append(f"- Link: {clean_text(link)}")

    # ---- Knowledge Graph ----
    knowledge_graph = pick(parsed, "knowledge_graph")
    if isinstance(knowledge_graph, dict):
        kg_lines = []

        for key in ["title", "type", "description"]:
            value = pick(knowledge_graph, key)
            if value:
                kg_lines.append(f"- {key}: {truncate(value, 500)}")

        if kg_lines:
            lines.append("\n## Knowledge Graph")
            lines.extend(kg_lines)

    # ---- 如果没有抓到主要内容，输出 compact JSON 摘要 ----
    final_text = "\n".join(lines).strip()

    if final_text == "# Google Search / SerpAPI 清洗结果":
        compact = json.dumps(parsed, ensure_ascii=False)
        compact = clean_text(compact)

        if not compact:
            return {
                "output": "No searchable content extracted. 请确认 input 接的是 Google / SerpAPI HTTP body 原始输出。"
            }

        return {
            "output": compact[:30000]
        }

    # 限制长度，避免下游爆 token
    final_text = final_text[:30000]

    return {
        "output": final_text
    }
